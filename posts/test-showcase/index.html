Code: CSS#

The custom palette override — Molokai colours applied to PaperMod’s CSS variables:

/* ─── Molokai-inspired terminal palette ─────────────────────── */
[data-theme="dark"] {
    --theme:          #1b1d1e;   /* molokai background         */
    --entry:          #272822;   /* molokai current-line       */
    --primary:        #f8f8f2;   /* molokai foreground         */
    --secondary:      #75715e;   /* molokai comments           */
    --tertiary:       #3e3d32;   /* slightly elevated bg       */
    --border:         #465457;   /* molokai line-number column */
}

Code: Python#

Parse Hugo’s index.json search index and print post titles with their URLs:

import json
from pathlib import Path


def load_search_index(public_dir: str = "public") -> list[dict]:
    index_path = Path(public_dir) / "index.json"
    with index_path.open() as f:
        return json.load(f)


def print_posts(entries: list[dict]) -> None:
    for entry in sorted(entries, key=lambda e: e.get("date", ""), reverse=True):
        title = entry.get("title", "(untitled)")
        url = entry.get("permalink", "")
        tags = ", ".join(entry.get("tags", []))
        print(f"{title}")
        print(f"  url  : {url}")
        if tags:
            print(f"  tags : {tags}")


if __name__ == "__main__":
    posts = load_search_index()
    print_posts(posts)
