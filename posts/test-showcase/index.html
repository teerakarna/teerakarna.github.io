Code: Python#

A quick AWS helper — list ECR image tags for a given repository:

import boto3

def list_image_tags(repo: str, registry_id: str = "406214174851") -> list[str]:
    ecr = boto3.client("ecr", region_name="eu-central-1")
    paginator = ecr.get_paginator("list_images")
    tags = []
    for page in paginator.paginate(registryId=registry_id, repositoryName=repo):
        tags += [
            img["imageTag"]
            for img in page["imageIds"]
            if "imageTag" in img
        ]
    return sorted(tags, reverse=True)

if __name__ == "__main__":
    for tag in list_image_tags("demo-service"):
        print(tag)
